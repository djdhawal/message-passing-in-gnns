\section{The question}

\begin{frame}{Message passing: a game of telephone}
  \centering
  \begin{tikzpicture}[n/.style={circle,draw=ink!70,fill=soft,minimum size=0.75cm,font=\small},
      arr/.style={-{Latex[length=2mm]},thick,accent}]
    \foreach \i/\l in {0/u,1/,2/,3/,4/v} \node[n] (p\i) at (\i*2.2,0) {$\l$};
    \fill[warm] (p0) circle (4pt);
    \foreach \i in {0,...,3} {
      \pgfmathtruncatemacro\j{\i+1}
      \draw[arr] (p\i) to[bend left=25] node[above,font=\scriptsize,ink] {layer \j} (p\j);
    }
    \node[below=0.15cm of p0,font=\scriptsize,warm] {the message};
    \node[below=0.15cm of p4,font=\scriptsize,ink] {needs it};
  \end{tikzpicture}
  \vspace{0.8em}
  \raggedright
  \begin{itemize}
    \item A graph neural network (GNN) updates each node from its \textbf{neighbours}
    \item One layer = one hop. With $L$ layers a node hears at most $L$ hops away
    \item Far information is \textbf{relayed} node by node, and mixed with everything else on the way
  \end{itemize}
\end{frame}

\begin{frame}{Over-squashing: many voices through one bridge}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      \centering
      \begin{tikzpicture}[n/.style={circle,fill=accent!70,inner sep=2.2pt},m/.style={circle,fill=good!80,inner sep=2.2pt}]
        \foreach \i in {0,...,5} \node[n] (a\i) at ($(0,0)+(\i*60:0.95)$) {};
        \foreach \i in {0,...,5} \node[m] (b\i) at ($(3.6,0)+(\i*60:0.95)$) {};
        \foreach \i in {0,...,4} {
          \pgfmathtruncatemacro\s{\i+1}
          \foreach \j in {\s,...,5} {\draw[ink!35] (a\i)--(a\j); \draw[ink!35] (b\i)--(b\j);}
        }
        \draw[warm,line width=2pt] (a0) -- (b3);
        \node[above,font=\scriptsize,warm] at (1.8,0.05) {bridge};
        \node[font=\scriptsize] at (0,-1.35) {cluster A};
        \node[font=\scriptsize] at (3.6,-1.35) {cluster B};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item Everything B knows must reach A through \textbf{one edge}
        \item Each node stores a fixed-size vector, so the messages get \textbf{compressed and lost}
        \item This is \alert{over-squashing} \snote{(Alon \& Yahav 2021; Topping et al.\ 2022)}
        \item A narrow bridge = high \textbf{effective resistance} between A and B
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{Measures}

\begin{frame}{Effective resistance collapses to degree}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \[ R(u,v) = (e_u - e_v)^\top L^+ (e_u - e_v) \]
      \begin{itemize}\small
        \item Low $R$: many wide paths. High $R$: a bottleneck
        \item On large, well-connected graphs \snote{(von Luxburg, Radl \& Hein 2014)}:
          \[ R(u,v) \;\to\; \tfrac{1}{d_u} + \tfrac{1}{d_v} \]
        \item So $R$ mostly measures \alert{degree}, not bottlenecks
        \item We use \textbf{excess resistance}
          \[ R_{\text{ex}}(u,v) = R(u,v) - \big(\tfrac{1}{d_u} + \tfrac{1}{d_v}\big) \]
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering\small
      \begin{tabular}{@{}lcc@{}}
        \toprule
        Graph & $\rho_S\big(R,\ \tfrac1{d_u}{+}\tfrac1{d_v}\big)$ & mean $R_{\text{ex}}$ \\
        \midrule
        twitch-gamers & 0.90 & 0.39 \\
        reddit & 0.77 & 0.49 \\
        twitch-de & \alert{0.998} & 0.02 \\
        higgs-follow & \alert{0.98} & 0.09 \\
        higgs-retweet & 0.07 & -- \\
        \bottomrule
      \end{tabular}

      \snote{$\rho_S$: Spearman correlation, on test subgraphs. Only the very sparse retweet layer escapes the collapse.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Three model measures}
  \begin{columns}[T]
    \begin{column}{0.32\textwidth}
      \textbf{Jacobian sensitivity}
      \[ J(u,v) = \Big\|\frac{\partial h_v^{(L)}}{\partial h_u^{(0)}}\Big\|_F \]
      \begin{itemize}\small
        \item Does $v$'s output move when $u$'s input moves?
        \item Over-squashing: $J$ \textbf{decays} with hops and resistance
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Attention entropy}
      \[ \tilde H = \frac{-\sum_j a_{ij}\log a_{ij}}{\log N} \in [0,1] \]
      \begin{itemize}\small
        \item $\tilde H \approx 1$: attention spread evenly over everyone
        \item High entropy = \textbf{dilution}
        \item Exact for softmax; implicit rows for Performer
      \end{itemize}
    \end{column}
    \begin{column}{0.32\textwidth}
      \textbf{Range}
      \[ \rho_v = \frac{\sum_u I_{vu}\, d(v,u)}{\sum_u I_{vu}} \]
      \begin{itemize}\small
        \item How far the trained model listens
        \item Node level, from the Jacobian \snote{(Bamberger et al.\ 2025)}
        \item Distance in hops or (excess) resistance
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{The knob problem}

\begin{frame}{The hybrid model and its $\alpha$ knob}
  \[ x' = x + (1-\alpha)\,\mathrm{Norm}\big(\mathrm{MPNN}(x)\big) + \alpha\,\mathrm{Norm}\big(\mathrm{Attn}(x)\big) \]
  \begin{itemize}
    \item $\alpha = 0$: pure message passing \quad $\alpha=1$: pure attention \snote{(GraphGPS-like, Rampášek et al.\ 2022)}
    \item Plan: sweep $\alpha$ and find the best mix per graph and task
  \end{itemize}
  \vspace{0.4em}
  \centering
  \begin{tikzpicture}
    \draw[-{Latex},thick,ink!60] (0,0) -- (9,0);
    \node[left,font=\small,align=right] at (0,0) {$\alpha=0$\\over-squashing};
    \node[right,font=\small,align=left] at (9,0) {$\alpha=1$\\dilution};
    \node[above,font=\small,accent] at (4.5,0.05) {hybrids};
    \fill[accent] (4.5,0) circle (2.5pt);
  \end{tikzpicture}
  \vspace{0.4em}

  \raggedright
  \alert{But}: as built, training could undo $\alpha$. Next slide.
\end{frame}

\begin{frame}{Why $\alpha$ did nothing: the norms absorb it}
  BatchNorm has a learnable scale $\gamma$:\quad $\mathrm{Norm}(h) = \gamma\,\hat h + \beta$
  \[ (1-\alpha)\,\mathrm{Norm}(h_{\text{loc}}) + \alpha\,\mathrm{Norm}(h_{\text{glb}})
     = \underbrace{(1-\alpha)\gamma_{\text{loc}}}_{\text{learned}}\,\hat h_{\text{loc}}
     + \underbrace{\alpha\,\gamma_{\text{glb}}}_{\text{learned}}\,\hat h_{\text{glb}} + \dots \]
  \begin{itemize}
    \item Training can rescale either branch freely
    \item Any $\alpha$ strictly between 0 and 1 gives the \textbf{same} model family
    \item $\alpha$ only changed the \alert{starting point}, not the model
  \end{itemize}
  \vspace{0.3em}
  \begin{exampleblock}{Fix 1: affine-free $\alpha$}
    Remove the learnable scale from the branch norms (\texttt{branch\_affine=false}). Now $\alpha$ is a real, fixed weight.
  \end{exampleblock}
\end{frame}

\begin{frame}{Fix 2: the attention radius $k$}
  \begin{columns}[c]
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}
        \fill[accent!8] (0,0) circle (1.9);
        \fill[accent!18] (0,0) circle (1.25);
        \fill[accent!32] (0,0) circle (0.6);
        \foreach \r in {0.6,1.25,1.9} \draw[accent!60] (0,0) circle (\r);
        \node[circle,fill=warm,inner sep=2.6pt] at (0,0) {};
        \node[font=\scriptsize] at (0,0.38) {$k{=}1$};
        \node[font=\scriptsize] at (0,0.95) {$k{=}2$};
        \node[font=\scriptsize] at (0,1.6) {$k{=}3$};
        \node[font=\scriptsize,ink!70] at (0,-2.2) {$k=\infty$: the whole graph};
      \end{tikzpicture}
    \end{column}
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item Attention may only look at nodes \textbf{within $k$ hops}
        \item $k=0$: pure message passing; $k=\infty$: full attention
        \item Larger $k$: shortcuts past bottlenecks (\textcolor{good}{less over-squashing})
        \item \ldots but more nodes per softmax (\alert{more dilution})
        \item The best $k$ \emph{is} the trade-off we study
        \item Learned weights \textbf{cannot undo} it
      \end{itemize}
      \snote{Related architectures: DRew (Gutteridge et al.\ 2023, $k$-hop rewiring), Shortest Path Networks (Abboud et al.\ 2022), Graphormer's distance bias (Ying et al.\ 2021). Our contribution: the radius as a \emph{diagnostic} knob, read with the measures.}
    \end{column}
  \end{columns}
\end{frame}

\section*{Summary}

\begin{frame}{Takeaways}
  \begin{itemize}
    \item GNNs pass messages hop by hop: far information is \textbf{squashed}; attention reaches far but \textbf{dilutes}
    \item Social graphs are small-world with hubs, and no long-range benchmark covers them
    \item Resistance collapses to degree there, so we use \textbf{excess resistance}
    \item The $\alpha$ knob could be undone by the norms: fixed (affine-free) and joined by the \textbf{attention radius $k$}
    \item On Higgs, no graph model beat the MLP yet: \alert{subgraph cuts removed most of the neighbour signal}
    \item Now: full-graph sweeps and planted targets with known range; then the phase diagram, GFMs and a benchmark
  \end{itemize}
\end{frame}

        \item Every hybrid solves it, and beats pure attention
        \item \alert{Confounded}: different target draws, mean aggregation for a sum target, attention-only hop bias, unequal parameter counts
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
